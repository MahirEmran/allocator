36static void* allocate(std::size_t size, std::size_t id = 0) {
37 if (size == 0) size = 1;
38 // Default to arena if slab couldn't work.
39 // We will want to always pass in id, since
40 // it could default to arena. And if we run
41 // in parallel it would be bad.
42 if (size <= HYBRID_THRESHOLD) {
43 void* p = slab::allocate(size);
44 if (p) return p;
45 }
46 if (id >= NUM_ARENAS) return nullptr; // GCOVR_EXCL_BR_LINE
47 return arenas[id].allocate(size);
48}
